                        WHEN CAST(TO_CHAR(SHIP_CONFIRM.SHIP_CONF,&apos;HH24&apos;) AS FLOAT) &gt;= 6 AND CAST(TO_CHAR(SHIP_CONFIRM.SHIP_CONF,&apos;HH24&apos;) AS FLOAT) &lt; 8 THEN &apos;1º - turno&apos;

                    END
                WHEN SH.SHIPMENT_STATUS=80 AND SHIP_CONFIRM.SHIP_CONF IS NULL THEN

                        WHEN CAST(TO_CHAR(SH.CREATED_DTTM,&apos;HH24&apos;) AS FLOAT) &gt;= 22 THEN &apos;3º - turno&apos;
